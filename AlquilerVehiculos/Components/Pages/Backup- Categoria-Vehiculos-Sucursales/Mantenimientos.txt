@code {
    private List<Mantenimiento> mantenimientos = new();
    private List<Vehiculo> vehiculos = new();

    private FormularioMantenimiento formulario = new();

    private bool cargando = true;
    private bool guardando;
    private bool mostrarFormulario;
    private bool esEdicion;

    private int? idEdicion;

    private string? mensaje;
    private string? error;

    protected override async Task OnInitializedAsync()
    {
        await CargarDatos();
    }

    private async Task CargarDatos()
    {
        cargando = true;

        try
        {
            await using var db = await DbFactory.CreateDbContextAsync();

            mantenimientos = await db.Mantenimiento
                .AsNoTracking()
                .Include(m => m.IdVehiculoNavigation)
                .OrderBy(m => m.IdMantenimiento)
                .ToListAsync();

            vehiculos = await db.Vehiculo
                .AsNoTracking()
                .OrderBy(v => v.IdVehiculo)
                .ToListAsync();
        }
        catch (Exception)
        {
            error = "No se pudieron cargar los datos del módulo.";
        }
        finally
        {
            cargando = false;
        }
    }

    private async Task NuevoMantenimiento()
    {
        error = null;
        mensaje = null;

        await CargarDatos();

        formulario = new FormularioMantenimiento
        {
            FechaInicio = DateOnly.FromDateTime(DateTime.Today),
            Estado = "en_curso"
        };

        idEdicion = null;
        esEdicion = false;
        mostrarFormulario = true;
    }

    private async Task Editar(int id)
    {
        error = null;
        mensaje = null;

        try
        {
            await using var db = await DbFactory.CreateDbContextAsync();

            var m = await db.Mantenimiento
                .AsNoTracking()
                .FirstOrDefaultAsync(x => x.IdMantenimiento == id);

            if (m is null)
            {
                error = "El mantenimiento ya no existe.";
                return;
            }

            formulario = new FormularioMantenimiento
            {
                IdVehiculo = m.IdVehiculo,
                Tipo = m.Tipo,
                Descripcion = m.Descripcion,
                Taller = m.Taller,
                FechaInicio = m.FechaInicio,
                FechaFin = m.FechaFin,
                Costo = m.Costo,
                Kilometraje = m.Kilometraje,
                Estado = m.Estado
            };

            idEdicion = id;
            esEdicion = true;
            mostrarFormulario = true;

            await CargarDatos();
        }
        catch (Exception)
        {
            error = "No se pudo cargar el mantenimiento.";
        }
    }

    private async Task Guardar()
    {
        if (guardando)
            return;

        guardando = true;
        error = null;
        mensaje = null;

        try
        {
            if (formulario.IdVehiculo <= 0)
            {
                error = "Debes seleccionar un vehículo.";
                return;
            }

            if (formulario.FechaFin.HasValue &&
                formulario.FechaFin.Value < formulario.FechaInicio)
            {
                error = "La fecha de finalización no puede ser anterior a la fecha de inicio.";
                return;
            }

            if (formulario.Estado == "completado" &&
                !formulario.FechaFin.HasValue)
            {
                error = "Debes indicar la fecha de finalización para completar el mantenimiento.";
                return;
            }

          if (formulario.Estado == "en_curso")
{
    formulario.FechaFin = null;
}
else if (!formulario.FechaFin.HasValue)
{
    error = "Debes indicar la fecha de finalización.";
    return;
}

            await using var db = await DbFactory.CreateDbContextAsync();
            await using var transaction = await db.Database.BeginTransactionAsync();

            var vehiculo = await db.Vehiculo
                .FirstOrDefaultAsync(v => v.IdVehiculo == formulario.IdVehiculo);

            if (vehiculo is null)
            {
                error = "El vehículo seleccionado no existe.";
                return;
            }

            bool yaTieneMantenimiento = await db.Mantenimiento.AnyAsync(m =>
                 m.IdVehiculo == formulario.IdVehiculo &&
                 m.Estado == "en_curso" &&
                (!esEdicion || m.IdMantenimiento != idEdicion));

            if (formulario.Estado == "en_curso" && yaTieneMantenimiento)
               {
                   error = "Este vehículo ya tiene un mantenimiento en curso. Finalízalo antes de registrar otro.";
                   return;
           }

            Mantenimiento mantenimiento;

            if (esEdicion && idEdicion.HasValue)
            {
                mantenimiento = await db.Mantenimiento
                    .FirstOrDefaultAsync(m => m.IdMantenimiento == idEdicion.Value)
                    ?? throw new InvalidOperationException("El mantenimiento ya no existe.");
            }
            else
            {
                mantenimiento = new Mantenimiento();
                db.Mantenimiento.Add(mantenimiento);
            }

            mantenimiento.IdVehiculo = formulario.IdVehiculo;
            mantenimiento.Tipo = formulario.Tipo;
            mantenimiento.Descripcion = formulario.Descripcion.Trim();
            mantenimiento.Taller = Limpiar(formulario.Taller);
            mantenimiento.FechaInicio = formulario.FechaInicio;
            mantenimiento.FechaFin = formulario.FechaFin;
            mantenimiento.Costo = formulario.Costo;
            mantenimiento.Kilometraje = formulario.Kilometraje;
            mantenimiento.Estado = formulario.Estado;

            if (formulario.Estado == "en_curso")
            {
                vehiculo.Estado = "en_mantenimiento";
            }
            else
            {
                bool tieneOtroMantenimiento = await db.Mantenimiento.AnyAsync(m =>
                    m.IdVehiculo == formulario.IdVehiculo &&
                    m.Estado == "en_curso" &&
                    (!esEdicion || m.IdMantenimiento != idEdicion));

                if (!tieneOtroMantenimiento &&
                    vehiculo.Estado == "en_mantenimiento")
                {
                    vehiculo.Estado = "activo";
                }
            }

            await db.SaveChangesAsync();
            await transaction.CommitAsync();

            mensaje = esEdicion
                ? "Mantenimiento actualizado correctamente."
                : "Mantenimiento registrado correctamente.";

            mostrarFormulario = false;
            await CargarDatos();
        }
        catch (DbUpdateException)
        {
            error = "No se pudo guardar el mantenimiento. Verifica los datos y las restricciones de la base de datos.";
        }
        catch (Exception)
        {
            error = "Ocurrió un error al guardar el mantenimiento.";
        }
        finally
        {
            guardando = false;
        }
    }

    private async Task CambiarEstado(int id)
    {
        if (guardando)
            return;

        guardando = true;
        error = null;
        mensaje = null;

        try
        {
            await using var db = await DbFactory.CreateDbContextAsync();
            await using var transaction = await db.Database.BeginTransactionAsync();

            var m = await db.Mantenimiento
                .FirstOrDefaultAsync(x => x.IdMantenimiento == id);

            if (m is null)
            {
                error = "El mantenimiento ya no existe.";
                return;
            }

            var vehiculo = await db.Vehiculo
                .FirstOrDefaultAsync(v => v.IdVehiculo == m.IdVehiculo);

            if (vehiculo is null)
            {
                error = "No se encontró el vehículo relacionado.";
                return;
            }

            if (m.Estado == "en_curso")
            {
                m.Estado = "completado";
                m.FechaFin = DateOnly.FromDateTime(DateTime.Today);

                bool otroPendiente = await db.Mantenimiento.AnyAsync(x =>
                    x.IdVehiculo == m.IdVehiculo &&
                    x.Estado == "en_curso" &&
                    x.IdMantenimiento != id);

                if (!otroPendiente && vehiculo.Estado == "en_mantenimiento")
                    vehiculo.Estado = "activo";

                mensaje = "Mantenimiento finalizado correctamente.";
            }
          

            await db.SaveChangesAsync();
            await transaction.CommitAsync();

            await CargarDatos();
        }
        catch (Exception)
        {
            error = "No se pudo cambiar el estado del mantenimiento.";
        }
        finally
        {
            guardando = false;
        }
    }

    private void Cancelar()
    {
        mostrarFormulario = false;
        formulario = new FormularioMantenimiento();
        idEdicion = null;
        esEdicion = false;
    }

    private static string? Limpiar(string? texto)
    {
        return string.IsNullOrWhiteSpace(texto) ? null : texto.Trim();
    }

    private sealed class FormularioMantenimiento
    {
        [Range(1, int.MaxValue, ErrorMessage = "Selecciona un vehículo.")]
        public int IdVehiculo { get; set; }

        [Required(ErrorMessage = "Selecciona el tipo de mantenimiento.")]
        public string Tipo { get; set; } = "";

        [Required(ErrorMessage = "La descripción es obligatoria.")]
        [StringLength(300, ErrorMessage = "Máximo 300 caracteres.")]
        public string Descripcion { get; set; } = "";

        [StringLength(100, ErrorMessage = "Máximo 100 caracteres.")]
        public string? Taller { get; set; }

        [Required]
        public DateOnly FechaInicio { get; set; }

        public DateOnly? FechaFin { get; set; }

        [Range(typeof(decimal), "0", "99999999.99",
            ErrorMessage = "El costo debe ser igual o mayor que cero.")]
        public decimal Costo { get; set; }

        [Range(0, int.MaxValue,
            ErrorMessage = "El kilometraje no puede ser negativo.")]
        public int? Kilometraje { get; set; }

        [Required]
        public string Estado { get; set; } = "en_curso";
    }
}
